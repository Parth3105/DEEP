@startuml

actor student

boundary homePageTemplate
boundary registrationTemplate

control "EC.REF" as EnrollmentController.renderEnrollmentForm
control "EC.LSP" as EnrollmentController.loadSubmittedPreferences
control "HPC.RSHP" as HomePageController.renderStudentHomePage
control "PARC.LMPS" as PreferenceAndResultController.loadMyPreferenceSummary

participant StudentService
participant CourseService
participant InstituteReqService
participant StudentReqService
participant CoursePrefService
participant SlotPrefService
participant SystemStatusService

participant StudentRepo
participant CourseRepo
participant InstituteReqRepo
participant StudentReqRepo
participant CoursePrefRepo
participant SlotPrefRepo
participant SystemStatusRepo


activate student

/' Rendering Preference Selection Page Part '/

student -> homePageTemplate : begin selection
activate homePageTemplate
homePageTemplate -> EnrollmentController.renderEnrollmentForm : GET /student/preference-form
activate EnrollmentController.renderEnrollmentForm

EnrollmentController.renderEnrollmentForm -> SystemStatusService : fetchRegistrationStatus()
activate SystemStatusService
SystemStatusService -> SystemStatusRepo : findById()
activate SystemStatusRepo
SystemStatusRepo --> SystemStatusService : return preference collection status 
deactivate SystemStatusRepo
SystemStatusService --> EnrollmentController.renderEnrollmentForm : return preference collection status
deactivate SystemStatusService

opt if preference collection status is not open 
  EnrollmentController.renderEnrollmentForm -> HomePageController.renderStudentHomePage : redirect:/student/home-page
  activate HomePageController.renderStudentHomePage
  note over HomePageController.renderStudentHomePage 
    HomePageController.renderStudentHomePage 
    continues in "Student HomePage - Sequence Diagram".
  end note
  deactivate HomePageController.renderStudentHomePage
end

EnrollmentController.renderEnrollmentForm -> StudentService : fetchEnrollementStatusForStudent()
activate StudentService
StudentService -> StudentRepo : findById()
activate StudentRepo
StudentRepo --> StudentService : return student info
deactivate StudentRepo
StudentService --> EnrollmentController.renderEnrollmentForm : return preference submission status 
deactivate StudentService

opt if student has submitted preferences 
  EnrollmentController.renderEnrollmentForm -> PreferenceAndResultController.loadMyPreferenceSummary : redirect:/student/my-preferences
  activate PreferenceAndResultController.loadMyPreferenceSummary
  note over PreferenceAndResultController.loadMyPreferenceSummary
    PreferenceAndResultController.loadMyPreferenceSummary
    continues in "Student HomePage - Sequence Diagram".
  end note
  deactivate PreferenceAndResultController.loadMyPreferenceSummary
end

EnrollmentController.renderEnrollmentForm -> StudentService : fetchStudentData()
activate StudentService
StudentService -> StudentRepo : findById()
activate StudentRepo
StudentRepo --> StudentService : return student info
deactivate StudentRepo
StudentService --> EnrollmentController.renderEnrollmentForm : return student info
deactivate StudentService

opt if student is null 
  EnrollmentController.renderEnrollmentForm -> HomePageController.renderStudentHomePage : redirect:/student/home-page
  activate HomePageController.renderStudentHomePage
  note over HomePageController.renderStudentHomePage 
    HomePageController.renderStudentHomePage 
    continues in "Student HomePage - Sequence Diagram".
  end note
  deactivate HomePageController.renderStudentHomePage
end

par 

  EnrollmentController.renderEnrollmentForm -> InstituteReqService : findInstituteReq()
  activate InstituteReqService
  InstituteReqService -> InstituteReqRepo : findByProgramAndSemester()
  activate InstituteReqRepo
  InstituteReqRepo --> InstituteReqService : return list of instituteReq
  deactivate InstituteReqRepo
  InstituteReqService --> EnrollmentController.renderEnrollmentForm : return list of instituteReq
  deactivate InstituteReqService
  
else 

  EnrollmentController.renderEnrollmentForm -> CourseService : fetchAvailableCourses()
  activate CourseService
  CourseService -> CourseRepo : fetchAvailableCourses()
  activate CourseRepo
  CourseRepo --> CourseService : return required information of courses
  deactivate CourseRepo
  CourseService --> EnrollmentController.renderEnrollmentForm : return required information of courses
  deactivate CourseService
  
end

EnrollmentController.renderEnrollmentForm -> registrationTemplate : loads preference collection form
deactivate EnrollmentController.renderEnrollmentForm
deactivate homePageTemplate
activate registrationTemplate








/' Preference Submission Part '/

student -> registrationTemplate : submit 
registrationTemplate -> EnrollmentController.loadSubmittedPreferences : POST /student/submit-preferences
activate EnrollmentController.loadSubmittedPreferences

EnrollmentController.loadSubmittedPreferences -> SystemStatusService : fetchRegistrationStatus()
activate SystemStatusService
SystemStatusService -> SystemStatusRepo : findById()
activate SystemStatusRepo
SystemStatusRepo --> SystemStatusService : return preference collection status 
deactivate SystemStatusRepo
SystemStatusService --> EnrollmentController.loadSubmittedPreferences : return preference collection status
deactivate SystemStatusService

opt if preference collection status is not open 
  EnrollmentController.loadSubmittedPreferences -> HomePageController.renderStudentHomePage : redirect:/student/home-page
  activate HomePageController.renderStudentHomePage
  note over HomePageController.renderStudentHomePage 
    HomePageController.renderStudentHomePage 
    continues in "Student HomePage - Sequence Diagram".
  end note
  deactivate HomePageController.renderStudentHomePage
end

par 

  EnrollmentController.loadSubmittedPreferences -> StudentReqService : insertAll()
  activate StudentReqService
  StudentReqService -> StudentReqRepo : saveAll()
  activate StudentReqRepo
  deactivate StudentReqRepo
  deactivate StudentReqService
  
else 

  EnrollmentController.loadSubmittedPreferences -> CoursePrefService : insertAll()
  activate CoursePrefService
  CoursePrefService -> CoursePrefRepo : saveAll()
  activate CoursePrefRepo
  deactivate CoursePrefRepo
  deactivate CoursePrefService
  
else 

  EnrollmentController.loadSubmittedPreferences -> SlotPrefService : insertAll()
  activate SlotPrefService
  SlotPrefService -> SlotPrefRepo : saveAll()
  activate SlotPrefRepo
  deactivate SlotPrefRepo
  deactivate SlotPrefService

end

deactivate registrationTemplate
EnrollmentController.loadSubmittedPreferences -> PreferenceAndResultController.loadMyPreferenceSummary : redirect:/student/my-preferences
deactivate EnrollmentController.loadSubmittedPreferences
activate PreferenceAndResultController.loadMyPreferenceSummary
note over PreferenceAndResultController.loadMyPreferenceSummary
  PreferenceAndResultController.loadMyPreferenceSummary
  continues in "Student HomePage - Sequence Diagram".
end note
deactivate PreferenceAndResultController.loadMyPreferenceSummary

@enduml
