@startuml
participant AllocationSystem
participant AllocationDataLoader
participant DataLoader

participant StudentReqService
participant SlotPrefService
participant CoursePrefService
participant CourseService
participant CourseOfferingService
participant InstituteReqService
participant AllocationResultService
participant AllocationReportService
participant SeatSummaryService
participant StudentService

participant StudentReqRepo
participant SlotPrefRepo
participant CoursePrefRepo
participant CourseRepo
participant CourseOfferingRepo
participant InstituteReqRepo
participant AllocationResultRepo
participant AllocationReportRepo
participant SeatSummaryRepo
participant StudentRepo

par
  
    AllocationSystem -> AllocationDataLoader : getStudentData()
    activate AllocationDataLoader
    
    par
    
      AllocationDataLoader -> StudentService : fetchStudentsBySemester()
      activate StudentService
      StudentService -> StudentRepo : findBySemester()
      activate StudentRepo
      StudentRepo --> StudentService : return list of students 
      deactivate StudentRepo
      StudentService --> AllocationDataLoader : return list of students
      deactivate StudentService
      
    else
    
      AllocationDataLoader -> StudentReqService : fetchAllStudentReqs()
      activate StudentReqService
      StudentReqService -> StudentReqRepo : findAll()
      activate StudentReqRepo
      StudentReqRepo --> StudentReqService : return list of student requirements
      deactivate StudentReqRepo
      StudentReqService --> AllocationDataLoader : return list of student requirements
      deactivate StudentReqService
      
    else
    
      AllocationDataLoader -> SlotPrefService : fetchAllSlotSortedByPref()
      activate SlotPrefService
      SlotPrefService -> SlotPrefRepo : findAll(Sort.by(sid, pref).ascending())
      activate SlotPrefRepo
      SlotPrefRepo --> SlotPrefService : return list of slot preferences
      deactivate SlotPrefRepo
      SlotPrefService --> AllocationDataLoader : return list of slot preferences
      deactivate SlotPrefService
      
    else
    
      AllocationDataLoader -> CoursePrefService : fetchAllCoursePrefSortedByPref()
      activate CoursePrefService
      CoursePrefService -> CoursePrefRepo : findAll(Sort.by(sid, pref).ascending())
      activate CoursePrefRepo
      CoursePrefRepo --> CoursePrefService : return list of slot preferences
      deactivate CoursePrefRepo
      CoursePrefService --> AllocationDataLoader : return list of slot preferences
      deactivate CoursePrefService
      
    end
    
    opt if student not found
      AllocationDataLoader --> AllocationSystem : return null 
    end
    
    AllocationDataLoader --> AllocationSystem : return list of student data and their preferences
    deactivate AllocationDataLoader

  else 
    
    AllocationSystem -> AllocationDataLoader : getCourseData()
    activate AllocationDataLoader
    AllocationDataLoader -> CourseService : fetchAllCourses()
    activate CourseService
    CourseService -> CourseRepo : findAll()
    activate CourseRepo
    CourseRepo --> CourseService : return list of courses
    deactivate CourseRepo
    CourseService --> AllocationDataLoader : return list of courses
    deactivate CourseService
    AllocationDataLoader --> AllocationSystem : return course data 
    deactivate AllocationDataLoader
    
  else 
  
    AllocationSystem -> AllocationDataLoader : getCourseOffers()
    activate AllocationDataLoader
    AllocationDataLoader -> CourseOfferingService : fetchCourseOfferingBySemester()
    activate CourseOfferingService
    CourseOfferingService -> CourseOfferingRepo : findBySemester()
    activate CourseOfferingRepo
    CourseOfferingRepo --> CourseOfferingService : return list of course offering 
    deactivate CourseOfferingRepo
    CourseOfferingService --> AllocationDataLoader : return list of course offering
    deactivate CourseOfferingService
    AllocationDataLoader --> AllocationSystem : return course offering data 
    deactivate AllocationDataLoader
    
  else
  
    AllocationSystem -> AllocationDataLoader : getInstituteRequirements()
    activate AllocationDataLoader
    AllocationDataLoader -> InstituteReqService : fetchInstituteReqBySemester()
    activate InstituteReqService
    InstituteReqService -> InstituteReqRepo : findBySemester()
    activate InstituteReqRepo
    InstituteReqRepo --> InstituteReqService : return list of instituteReqs
    deactivate InstituteReqRepo
    InstituteReqService --> AllocationDataLoader : return list of instituteReqs
    deactivate InstituteReqService
    AllocationDataLoader --> AllocationSystem : return institute requirements 
    deactivate AllocationDataLoader
    
  end
  
  AllocationSystem -> AllocationSystem : allocationInPhase() 
  activate AllocationSystem
    loop i = 0 to 1
      AllocationSystem -> AllocationSystem : allocationPhase((i == 0))
      activate AllocationSystem
        loop i = 0 to maxRequirements[0]
          AllocationSystem -> AllocationSystem: updatePriorityGroups()
          activate AllocationSystem
          deactivate AllocationSystem
            
          loop for each priority in priorities
            loop for each student in priorityGroups.get(priority)
              AllocationSystem -> AllocationSystem : courseAllocation(student, isPhaseOne, false)
              activate AllocationSystem
                loop prefIndex = 0 to maxPrefIndex - 1
                  loop for each slotPref in slotPrefs
                    AllocationSystem -> AllocationSystem : canAllocateCourse(student, courseID, isPhaseOne, isErrorPhase)
                    activate AllocationSystem
                      AllocationSystem --> AllocationSystem : return true / false
                    deactivate AllocationSystem
                    break if canAllocate
                      AllocationSystem -> AllocationSystem : allocateCourse(student, courseID)
                      activate AllocationSystem
                      deactivate AllocationSystem
                    end
                  end
                  
                  break if isAllocated
                    AllocationSystem --> AllocationSystem : return allocated
                  end
                end
              deactivate AllocationSystem
            end
          end
        end
      deactivate AllocationSystem
      AllocationSystem -> AllocationSystem : isStudentReqFulfilled
      activate AllocationSystem
        AllocationSystem --> AllocationSystem : return true / false
      deactivate AllocationSystem
    end
  
  AllocationSystem -> AllocationSystem : saveOutput()
  activate AllocationSystem
  
  par 

    AllocationSystem -> AllocationDataLoader : saveAllocationResult()
    activate AllocationDataLoader
    deactivate AllocationDataLoader
    AllocationSystem -> DataLoader : createCourseWiseAllocation()
    activate DataLoader
    
    loop for each course in courses  
      DataLoader -> AllocationResultService : fetchCourseWiseAllocation()
      activate AllocationResultService
      AllocationResultService -> AllocationResultRepo : findByCidOrderBySid()
      activate AllocationResultRepo
      deactivate AllocationResultRepo
      deactivate AllocationResultService
      DataLoader -> DataLoader : addToZip()
      activate DataLoader
      deactivate DataLoader
    end
    DataLoader --> AllocationSystem : return byte stream of file 
    deactivate DataLoader
    
    opt if byte stream is not null
      AllocationSystem -> AllocationReportService : insertReport() 
      activate AllocationReportService
      AllocationReportService -> AllocationReportService : deleteReport()
      activate AllocationReportService
      deactivate AllocationReportService
      AllocationReportService -> AllocationReportRepo : save() 
      activate AllocationReportRepo
      deactivate AllocationReportRepo
      deactivate AllocationReportService
    end
    
  else 
  
    AllocationSystem -> AllocationDataLoader : saveSeatSummary()
    activate AllocationDataLoader
    AllocationDataLoader -> SeatSummaryService : insertAll()
    activate SeatSummaryService
    SeatSummaryService -> SeatSummaryService : deleteAll()
    activate SeatSummaryService
    deactivate SeatSummaryService
    SeatSummaryService -> SeatSummaryRepo : saveAll()
    activate SeatSummaryRepo
    deactivate SeatSummaryRepo
    deactivate SeatSummaryService
    deactivate AllocationDataLoader
  
  else
  
    AllocationSystem -> DataLoader : createResultSheet()
    activate DataLoader
    DataLoader --> AllocationSystem : return byte stream of result sheet 
    deactivate DataLoader
    AllocationSystem -> AllocationReportService : insertReport() 
    activate AllocationReportService
    AllocationReportService -> AllocationReportService : deleteReport()
    activate AllocationReportService
    deactivate AllocationReportService
    AllocationReportService -> AllocationReportRepo : save() 
    activate AllocationReportRepo
    deactivate AllocationReportRepo
    deactivate AllocationReportService
  
  else 
  
    AllocationSystem -> DataLoader : createSeatSummary()
    activate DataLoader
    DataLoader --> AllocationSystem : return byte stream of seat summary 
    deactivate DataLoader
    AllocationSystem -> AllocationReportService : insertReport() 
    activate AllocationReportService
    AllocationReportService -> AllocationReportService : deleteReport()
    activate AllocationReportService
    deactivate AllocationReportService
    AllocationReportService -> AllocationReportRepo : save() 
    activate AllocationReportRepo
    deactivate AllocationReportRepo
    deactivate AllocationReportService
    
  else
  
    AllocationSystem -> AllocationSystem : getAllocationFailureDetail()
    activate AllocationSystem
    AllocationSystem --> AllocationSystem : return byte stream of failure allocation
    deactivate AllocationSystem
    AllocationSystem -> AllocationReportService : insertReport() 
    activate AllocationReportService
    AllocationReportService -> AllocationReportService : deleteReport()
    activate AllocationReportService
    deactivate AllocationReportService
    AllocationReportService -> AllocationReportRepo : save() 
    activate AllocationReportRepo
    deactivate AllocationReportRepo
    deactivate AllocationReportService
  
  end
  
  deactivate AllocationSystem
  
  @enduml
