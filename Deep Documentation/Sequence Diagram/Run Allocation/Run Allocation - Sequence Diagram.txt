@startuml

actor admin

boundary dashboardTemplate
boundary runAllocationTemplate

control "ASC.RRAP" as AllocationSystemController.renderRunAllocationPage
control "ASC.IA" as AllocationSystemController.initiateAllocation

participant StudentService
participant SystemStatusService
participant AllocationStatusService

participant StudentRepo
participant SystemStatusRepo
participant AllocationStatusRepo

participant AllocationSystem
participant RegistrationTaskManager


activate admin 



/' Rendering Run Allocation Page Part '/

admin -> dashboardTemplate : run allocation 
activate dashboardTemplate
dashboardTemplate -> AllocationSystemController.renderRunAllocationPage : GET /admin/run-allocation
activate AllocationSystemController.renderRunAllocationPage

par 

  AllocationSystemController.renderRunAllocationPage -> SystemStatusService : fetchRegistrationStatus()
  activate SystemStatusService
  SystemStatusService -> SystemStatusRepo : findById()
  activate SystemStatusRepo
  SystemStatusRepo --> SystemStatusService : return preference collection status
  deactivate SystemStatusRepo
  SystemStatusService --> AllocationSystemController.renderRunAllocationPage : return preference collection status
  deactivate SystemStatusService
  
else

 AllocationSystemController.renderRunAllocationPage -> AllocationStatusService : fetchAll()
 activate AllocationStatusService
 AllocationStatusService -> AllocationStatusRepo : findAll()
 activate AllocationStatusRepo
 AllocationStatusRepo --> AllocationStatusService : return list of allocation status 
 deactivate AllocationStatusRepo
 AllocationStatusService --> AllocationSystemController.renderRunAllocationPage : return list of allocation status
 deactivate AllocationStatusService
 
else

 AllocationSystemController.renderRunAllocationPage -> SystemStatusService : fetchResultStatus()
 activate SystemStatusService
 SystemStatusService -> SystemStatusRepo : findById()
 activate SystemStatusRepo
 SystemStatusRepo --> SystemStatusService : return result status
 deactivate SystemStatusRepo
 SystemStatusService --> AllocationSystemController.renderRunAllocationPage : return result status
 deactivate SystemStatusService
 
end

AllocationSystemController.renderRunAllocationPage --> runAllocationTemplate : loads run allocation page 
deactivate dashboardTemplate
deactivate AllocationSystemController.renderRunAllocationPage
activate runAllocationTemplate






/' Executing Run Allocation Page Part '/

admin -> runAllocationTemplate : execute allocation 
runAllocationTemplate -> AllocationSystemController.initiateAllocation : POST /admin/execute-allocation/{semester}
activate AllocationSystemController.initiateAllocation

par 

  opt if preference collection not ended
    AllocationSystemController.initiateAllocation -> SystemStatusService : updateOnClosingRegistration()
    activate SystemStatusService
    SystemStatusService -> RegistrationTaskManager : closeRegistration()
    activate RegistrationTaskManager
    RegistrationTaskManager -> SystemStatusService : autoCloseRegistration()
    deactivate RegistrationTaskManager
    SystemStatusService -> SystemStatusRepo : save()
    activate SystemStatusRepo
    deactivate SystemStatusRepo
    deactivate SystemStatusService
  end
  
else 

  AllocationSystemController.initiateAllocation -> AllocationSystem : initiateAllocation()
  activate AllocationSystem
  
  note over AllocationSystem
    The rest of the interaction of allocation system specifically is included in "Allocation System - Sequence Diagram"
    See that file for allocation system interaction with other participants.
  end note
  
  AllocationSystem --> AllocationSystemController.initiateAllocation : return suitable response
  deactivate AllocationSystem
  
  AllocationSystemController.initiateAllocation -> StudentService : countAllStudents()
  activate StudentService
  StudentService -> StudentRepo : count()
  activate StudentRepo
  StudentRepo --> StudentService : return number of students 
  deactivate StudentRepo
  StudentService --> AllocationSystemController.initiateAllocation : return number of students
  deactivate StudentService
  
  AllocationSystemController.initiateAllocation -> AllocationStatusService : insertAllocationStatus()
  activate AllocationStatusService
  AllocationStatusService -> AllocationStatusRepo : save()
  activate AllocationStatusRepo
  deactivate AllocationStatusRepo
  deactivate AllocationStatusService
  AllocationSystemController.initiateAllocation -> AllocationSystemController.renderRunAllocationPage : redirect:/admin/run-allocation 
  deactivate AllocationSystemController.initiateAllocation
  activate AllocationSystemController.renderRunAllocationPage
  
end
@enduml
