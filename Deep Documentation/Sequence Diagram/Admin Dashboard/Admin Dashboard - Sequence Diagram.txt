@startuml

actor admin

boundary dashboardTemplate

control "DC.OR" as DashboardController.openRegistration
control "DC.ERP" as DashboardController.extendRegistrationPeriod
control "DC.CR" as DashboardController.closeRegistration
control "DC.DR" as DashboardController.declareResults
control "NC.RDP" as NavigationController.renderDashboardPage

participant SystemStatusService

participant SystemStatusRepo

participant RegistrationtaskManager


activate admin


/' Begin Preference Collection Part '/

admin -> dashboardTemplate : begin preference collection 
activate dashboardTemplate
dashboardTemplate -> DashboardController.openRegistration : POST /admin/begin-collection
activate DashboardController.openRegistration

DashboardController.openRegistration -> SystemStatusService : updateOnOpeningRegistration()
activate SystemStatusService
SystemStatusService -> SystemStatusRepo : saveAll()
activate SystemStatusRepo
deactivate SystemStatusRepo

SystemStatusService -> RegistrationtaskManager : updateCloseRegistrationDate()
activate RegistrationtaskManager
deactivate RegistrationtaskManager
SystemStatusService -> RegistrationtaskManager : startRegistration()
activate RegistrationtaskManager
deactivate RegistrationtaskManager
deactivate SystemStatusService

DashboardController.openRegistration -> NavigationController.renderDashboardPage : redirect:/admin/admin-dashboard
deactivate DashboardController.openRegistration
deactivate dashboardTemplate
activate NavigationController.renderDashboardPage

note over NavigationController.renderDashboardPage
 NavigationController.renderDashboardPage 
 continues in "Admin - Navigation - Sequence Diagram".
end note






/' Extend Preference Collection Period Part '/

admin -> dashboardTemplate : extend preference collection period  
activate dashboardTemplate
deactivate NavigationController.renderDashboardPage
dashboardTemplate -> DashboardController.extendRegistrationPeriod : POST /admin/extend-period
activate DashboardController.extendRegistrationPeriod

DashboardController.extendRegistrationPeriod -> SystemStatusService : updateOnExtendingRegistrationPeriod()
activate SystemStatusService

alt if new date is before current date 
  SystemStatusService -> RegistrationtaskManager : closeRegistration()
  activate RegistrationtaskManager
  RegistrationtaskManager -> SystemStatusService : autoCloseRegistration()
  deactivate RegistrationtaskManager
  SystemStatusService -> SystemStatusRepo : save()
  activate SystemStatusRepo
  deactivate SystemStatusRepo
else 
  SystemStatusService -> RegistrationtaskManager : updateCloseRegistrationDate()
end

SystemStatusService -> SystemStatusRepo : save()
activate SystemStatusRepo
deactivate SystemStatusRepo
deactivate SystemStatusService
DashboardController.extendRegistrationPeriod -> NavigationController.renderDashboardPage : redirect:/admin/admin-dashboard
deactivate DashboardController.extendRegistrationPeriod
deactivate dashboardTemplate
activate NavigationController.renderDashboardPage

note over NavigationController.renderDashboardPage
 NavigationController.renderDashboardPage 
 continues in "Admin - Navigation - Sequence Diagram".
end note







/' End Preference Collection Part '/

admin -> dashboardTemplate : end preference collection  
activate dashboardTemplate
deactivate NavigationController.renderDashboardPage
dashboardTemplate -> DashboardController.closeRegistration : POST /admin/end-collection
activate DashboardController.closeRegistration

DashboardController.closeRegistration -> SystemStatusService : updateOnClosingRegistration()
activate SystemStatusService

SystemStatusService -> RegistrationtaskManager : closeRegistration()
activate RegistrationtaskManager
RegistrationtaskManager -> SystemStatusService : autoCloseRegistration()
deactivate RegistrationtaskManager
SystemStatusService -> SystemStatusRepo : save()
activate SystemStatusRepo
deactivate SystemStatusRepo
deactivate SystemStatusService

DashboardController.closeRegistration -> NavigationController.renderDashboardPage : redirect:/admin/admin-dashboard
deactivate DashboardController.closeRegistration
deactivate dashboardTemplate
activate NavigationController.renderDashboardPage

note over NavigationController.renderDashboardPage
 NavigationController.renderDashboardPage 
 continues in "Admin - Navigation - Sequence Diagram".
end note









/' Declare Result Part '/

admin -> dashboardTemplate : declare result  
activate dashboardTemplate
deactivate NavigationController.renderDashboardPage
dashboardTemplate -> DashboardController.declareResults : POST /admin/declare-results
activate DashboardController.declareResults

DashboardController.declareResults -> SystemStatusService : updateOnDeclaringResults()
activate SystemStatusService
SystemStatusService -> SystemStatusRepo : saveAll()
activate SystemStatusRepo
deactivate SystemStatusRepo
deactivate SystemStatusService

DashboardController.declareResults -> NavigationController.renderDashboardPage : redirect:/admin/admin-dashboard
deactivate DashboardController.declareResults
deactivate dashboardTemplate
activate NavigationController.renderDashboardPage

note over NavigationController.renderDashboardPage
 NavigationController.renderDashboardPage 
 continues in "Admin - Navigation - Sequence Diagram".
end note
deactivate NavigationController.renderDashboardPage


@enduml
