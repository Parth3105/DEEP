@startuml

actor admin 

boundary dashboardTemplate
boundary downloadReportsTemplate
boundary studentPreferencesTemplate
boundary allocationResultsTemplate
boundary adminFaqsTemplate

control NC.RDP as NavigationController.renderDashboardPage
control NC.RDRP as NavigationController.renderDownloadReportsPage
control NC.RSPP as NavigationController.renderStudentPreferencesPage
control NC.RARP as NavigationController.renderAllocationResultsPage
control NC.RFAQP as NavigationController.renderFaqPage

participant SystemStatusService
participant StudentService
participant AllocationStatusService

participant SystemStatusRepo
participant StudentRepo
participant AllocationStatusRepo

activate admin

' Rendering Admin DashBoard Page Part '

admin - NavigationController.renderDashboardPage  on login [GET adminadmin-dashboard]
activate NavigationController.renderDashboardPage

par 
  
  NavigationController.renderDashboardPage - SystemStatusService  fetchAllStatus()
  activate SystemStatusService
  SystemStatusService - SystemStatusRepo  findAll()
  activate SystemStatusRepo
  SystemStatusRepo -- SystemStatusService  return list of system status 
  deactivate SystemStatusRepo
  SystemStatusService -- NavigationController.renderDashboardPage  return list of system status 
  deactivate SystemStatusService
  
else 

  loop sem = 5 to 8
    par
    
      NavigationController.renderDashboardPage - StudentService  countBySemester()
      activate StudentService
      StudentService - StudentRepo  countBySemester()
      activate StudentRepo
      StudentRepo -- StudentService  return number of students 
      deactivate StudentRepo
      StudentService -- NavigationController.renderDashboardPage  return number of studentsd
      deactivate StudentService
      
    else 
    
      NavigationController.renderDashboardPage - StudentService  countEnrolledStudents()
      activate StudentService
      StudentService - StudentRepo  countByHasEnrolled()
      activate StudentRepo
      StudentRepo -- StudentService  return number of students 
      deactivate StudentRepo
      StudentService -- NavigationController.renderDashboardPage  return number of students
      deactivate StudentService
      
    else 
      
      NavigationController.renderDashboardPage - AllocationStatusService  checkIfExists()
      activate AllocationStatusService
      AllocationStatusService - AllocationStatusRepo  existsById()
      activate AllocationStatusRepo
      AllocationStatusRepo -- AllocationStatusService  return allocation status 
      deactivate AllocationStatusRepo
      AllocationStatusService -- NavigationController.renderDashboardPage  return allocation status
      deactivate AllocationStatusService

    end
  end
  
  NavigationController.renderDashboardPage -- dashboardTemplate  loads admin dashboard page 
  deactivate NavigationController.renderDashboardPage
  activate dashboardTemplate
end








' Rendering Download Reports Page Part '

admin - dashboardTemplate  download reports 
dashboardTemplate - NavigationController.renderDownloadReportsPage  GET admindownload-reports
activate NavigationController.renderDownloadReportsPage

NavigationController.renderDownloadReportsPage -- downloadReportsTemplate  loads download reports page 
deactivate NavigationController.renderDownloadReportsPage
deactivate dashboardTemplate
activate downloadReportsTemplate





' Rendering Student Preferences Page Part '
admin - dashboardTemplate  view preferences  student preferences 
deactivate downloadReportsTemplate
activate dashboardTemplate
dashboardTemplate - NavigationController.renderStudentPreferencesPage  GET adminstudent-preferences
activate NavigationController.renderStudentPreferencesPage

NavigationController.renderStudentPreferencesPage -- studentPreferencesTemplate  loads student preferences page 
deactivate NavigationController.renderStudentPreferencesPage
deactivate dashboardTemplate
activate studentPreferencesTemplate





' Rendering Allocation Result Page Part '

admin - dashboardTemplate  view allocation results  allocation result  
deactivate studentPreferencesTemplate
activate dashboardTemplate
dashboardTemplate - NavigationController.renderAllocationResultsPage  GET adminallocation-results
activate NavigationController.renderAllocationResultsPage

NavigationController.renderAllocationResultsPage -- allocationResultsTemplate  loads allocation result page 
deactivate NavigationController.renderAllocationResultsPage
deactivate dashboardTemplate
activate allocationResultsTemplate





' Rendering FAQs Page Part '

admin - dashboardTemplate  FAQs  
deactivate allocationResultsTemplate
activate dashboardTemplate
dashboardTemplate - NavigationController.renderFaqPage  GET adminfaqs
activate NavigationController.renderFaqPage

NavigationController.renderFaqPage -- adminFaqsTemplate  loads FAQs page 
deactivate NavigationController.renderFaqPage
deactivate dashboardTemplate
activate adminFaqsTemplate
deactivate adminFaqsTemplate
@enduml
