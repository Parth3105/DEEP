@startuml

actor user

boundary loginTemplate
boundary forgotPasswordTemplate
boundary verifyOtpTemplate
boundary resetPasswordTemplate
boundary homePageTemplate
boundary dashboardTemplate

participant SecurityFilterChain
participant AuthenticationSuccessHandler

control "LC.RLP" as LoginController.renderLoginPage
control "LC.ATH" as LoginController.authenticate
control "LC.RFPP" as LoginController.renderForgotPasswordPage
control "LC.ROTPVP" as LoginController.renderOtpVerificationPage
control "LC.LSID" as LoginController.loadStudentId
control "LC.ROTP" as LoginController.resendOtp
control "LC.LOTP" as LoginController.loadOtp
control "LC.LNP" as LoginController.loadNewPassword
control "NC.RDP" as NavigationController.renderDashboardPage
control "HPC.RSHP" as HomePageController.renderStudentHomePage

participant UserService
participant OtpVerificationService
participant EmailService

participant UserRepo
participant OtpVerificationRepo





activate user

  /' ------------------ Login Part ------------------ '/
  
  /' Requests Login Page '/
  user -> LoginController.renderLoginPage : GET /login
  activate LoginController.renderLoginPage
    LoginController.renderLoginPage --> loginTemplate : loads Login Page  
  deactivate LoginController.renderLoginPage
  
  /' Login through credentials '/
  activate loginTemplate
    user -> loginTemplate : sign in
    loginTemplate -> SecurityFilterChain : POST /authenticate
    activate SecurityFilterChain
    
      alt on correct credentials 
        SecurityFilterChain -> AuthenticationSuccessHandler : onAuthenticationSuccess()
       activate AuthenticationSuccessHandler
          alt if role = 'admin'
            AuthenticationSuccessHandler -> NavigationController.renderDashboardPage : GET /admin/admin-dashboard
            activate NavigationController.renderDashboardPage
              NavigationController.renderDashboardPage --> dashboardTemplate : loads admin dashboard
            deactivate NavigationController.renderDashboardPage
            activate dashboardTemplate
          else 
            AuthenticationSuccessHandler -> HomePageController.renderStudentHomePage : GET /student/home-page
            activate HomePageController.renderStudentHomePage
            HomePageController.renderStudentHomePage --> homePageTemplate : loads student home-page
            activate homePageTemplate
            deactivate HomePageController.renderStudentHomePage
          end
        deactivate AuthenticationSuccessHandler
      else 
        SecurityFilterChain --> loginTemplate : responds with suitable error
      end
    deactivate SecurityFilterChain
  deactivate loginTemplate
  deactivate dashboardTemplate
  deactivate homePageTemplate
  
  
  
  
  
  /' ------------------ Forgot Password Part ------------------ '/
  
  user -> loginTemplate : forgot password
  activate loginTemplate
    loginTemplate -> LoginController.renderForgotPasswordPage : GET /forgot-password
    activate LoginController.renderForgotPasswordPage
      LoginController.renderForgotPasswordPage --> forgotPasswordTemplate : loads Forgot Password Page
    deactivate LoginController.renderForgotPasswordPage
  deactivate loginTemplate
  
  user -> forgotPasswordTemplate : send OTP request
  activate forgotPasswordTemplate
  forgotPasswordTemplate -> LoginController.loadStudentId : POST /forgot-password
  activate LoginController.loadStudentId
  
  opt if session expires
    LoginController.loadStudentId -> LoginController.renderLoginPage : redirect:/login
    activate LoginController.renderLoginPage
      LoginController.renderLoginPage -> loginTemplate : loads login Page
    deactivate LoginController.renderLoginPage
    activate loginTemplate
  end
  
  activate LoginController.renderLoginPage
    LoginController.loadStudentId -> UserService : findUser()
    activate UserService
      UserService -> UserRepo : findById()
      activate UserRepo
        UserRepo --> UserService : return user
      deactivate UserRepo
      UserService --> LoginController.loadStudentId : return user
    deactivate UserService
    
    opt if user not found
      LoginController.loadStudentId -> LoginController.renderForgotPasswordPage : redirect:/forgot-password
      activate LoginController.renderForgotPasswordPage
        LoginController.renderForgotPasswordPage -> forgotPasswordTemplate : loads forgot-password Page
      deactivate LoginController.renderForgotPasswordPage
    end
    
    activate forgotPasswordTemplate
      LoginController.loadStudentId -> OtpVerificationService : generateOtpAndSendMail()
      activate OtpVerificationService
        OtpVerificationService -> OtpVerificationService : generateOtp()
        activate OtpVerificationService
        deactivate OtpVerificationService
        OtpVerificationService -> OtpVerificationRepo : save()
        activate OtpVerificationRepo
        deactivate OtpVerificationRepo
        OtpVerificationService -> EmailService : sendOtp()
        activate EmailService
        deactivate EmailService
      deactivate OtpVerificationService
      LoginController.loadStudentId -> LoginController.renderOtpVerificationPage : redirect:/verify-otp
      deactivate LoginController.loadStudentId
      activate LoginController.renderOtpVerificationPage
        LoginController.renderOtpVerificationPage --> verifyOtpTemplate : loads verify-otp Page 
      deactivate LoginController.renderOtpVerificationPage
      activate verifyOtpTemplate
    deactivate forgotPasswordTemplate
  deactivate LoginController.renderLoginPage
  deactivate loginTemplate
  
  
  
  
  
  
  /' ------------------ Resend OTP Part ------------------ '/
  
  user -> verifyOtpTemplate : re-send otp
  verifyOtpTemplate -> LoginController.resendOtp : POST /resend-otp
  activate LoginController.resendOtp
  
  opt if session expires
    LoginController.resendOtp -> LoginController.renderLoginPage : redirect:/login
    activate LoginController.renderLoginPage
      LoginController.renderLoginPage -> loginTemplate : loads login Page
    deactivate LoginController.renderLoginPage
    activate loginTemplate
  end
  
    LoginController.resendOtp -> OtpVerificationService : generateOtpAndSendMail()
    activate OtpVerificationService
      OtpVerificationService -> OtpVerificationService : generateOtp()
      activate OtpVerificationService
      deactivate OtpVerificationService
      OtpVerificationService -> OtpVerificationRepo : save()
      activate OtpVerificationRepo
      deactivate OtpVerificationRepo
      OtpVerificationService -> EmailService : sendOtp()
      activate EmailService
      deactivate EmailService
    deactivate OtpVerificationService
    LoginController.resendOtp -> LoginController.renderOtpVerificationPage : redirect:/verify-otp
    deactivate LoginController.resendOtp
    activate LoginController.renderOtpVerificationPage
      LoginController.renderOtpVerificationPage --> verifyOtpTemplate : loads verify-otp Page 
    deactivate LoginController.renderOtpVerificationPage
  deactivate loginTemplate
  
  
  
  
  
  
  /' ------------------ OTP Verification Part ------------------ '/
  
  user -> verifyOtpTemplate : verify-otp
  verifyOtpTemplate -> LoginController.loadOtp : POST /verify-otp
  activate LoginController.loadOtp
  
  opt if session expires
    LoginController.loadOtp -> LoginController.renderLoginPage : redirect:/login
    activate LoginController.renderLoginPage
      LoginController.renderLoginPage -> loginTemplate : loads login Page
    deactivate LoginController.renderLoginPage
    activate loginTemplate
  end
  
    LoginController.loadOtp -> OtpVerificationService : verifyOtp()
    activate OtpVerificationService
      OtpVerificationService -> OtpVerificationRepo : findById()
      activate OtpVerificationRepo
        OtpVerificationRepo --> OtpVerificationService : return otp information
      deactivate OtpVerificationRepo
      OtpVerificationService --> LoginController.loadOtp : return response
    deactivate OtpVerificationService
    
    opt if OTP expires or incorrect OTP 
      LoginController.loadOtp -> LoginController.renderOtpVerificationPage : redirect:/verify-otp
      activate LoginController.renderOtpVerificationPage
        renderOtpVerificationPage --> verifyOtpTemplate : loads verify-otp Page 
      deactivate LoginController.renderOtpVerificationPage
    end
    
    LoginController.loadOtp --> resetPasswordTemplate : loads reset password Page
    deactivate LoginController.loadOtp
    deactivate verifyOtpTemplate
    activate resetPasswordTemplate
  deactivate loginTemplate
  
  
  
  
  
  
  /' ------------------ Reset password Part ------------------ '/
  
  user -> resetPasswordTemplate : reset password
  resetPasswordTemplate -> LoginController.loadNewPassword : POST /reset-password
  activate LoginController.loadNewPassword
  
  opt if session expires
    LoginController.loadNewPassword -> LoginController.renderLoginPage : redirect:/login
    activate LoginController.renderLoginPage
      LoginController.renderLoginPage -> loginTemplate : loads login Page
    deactivate LoginController.renderLoginPage
    activate loginTemplate
  end
  
  LoginController.loadNewPassword -> UserService : resetPassword()
  activate UserService
    UserService -> UserRepo : updatePassword()
    activate UserRepo
    deactivate UserRepo
  deactivate UserService
  LoginController.loadNewPassword -> LoginController.renderLoginPage : redirect:/login
  deactivate resetPasswordTemplate
  activate LoginController.renderLoginPage
  deactivate LoginController.loadNewPassword
  LoginController.renderLoginPage -> loginTemplate : loads login Page
  deactivate LoginController.renderLoginPage
  deactivate loginTemplate
  
deactivate user
@enduml
